#!/usr/sbin/nft -f

flush ruleset

table inet filter {

chain input {
    type filter hook input priority 0; policy drop;

    # Loopback
    iif "lo" accept

    # Conexiones existentes
    ct state established,related accept

    # Tráfico inválido
    ct state invalid drop

    # Bloquear nuevas conexiones TCP cuando una IP
    # supera 10 SYN por segundo, con una ráfaga inicial de 20
    ct state new tcp flags syn \
        meter flood { ip saddr limit rate over 10/second burst 20 packets } \
        log prefix "NF-RATE-LIMIT: " counter drop

    # Servicios permitidos
    tcp dport 7777 accept
    tcp dport 80 accept
    tcp dport 443 accept

    # HTTP/3 / QUIC
    udp dport 443 accept

    # ICMP IPv4
    icmp type {
        echo-request,
        echo-reply,
        destination-unreachable,
        time-exceeded,
        parameter-problem
    } accept

    # ICMPv6
    icmpv6 type {
        echo-request,
        echo-reply,
        destination-unreachable,
        time-exceeded,
        parameter-problem,
        nd-router-solicit,
        nd-router-advert,
        nd-neighbor-solicit,
        nd-neighbor-advert
    } accept

    # Todo lo demás se registra y se contabiliza
    log prefix "NF-DROP: " limit rate 5/minute
    counter drop
}

chain forward {
    type filter hook forward priority 0; policy drop;

    counter drop
}

chain output {
    type filter hook output priority 0; policy accept;
}

}
